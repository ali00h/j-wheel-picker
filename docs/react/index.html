کد استفاده در React

نمونهٔ پایه با ref برای ورودی و پاک‌سازی نمونه هنگام unmount شدن کامپوننت.

import { useEffect, useRef, useState } from 'react';

function DateField() {
  const input = useRef(null);
  const [date, setDate] = useState('');

  useEffect(() => {
    const picker = new JWheelPicker({
      target: input.current,
      onSelect: value => setDate(value.formatted)
    });
    return () => picker.destroy();
  }, []);

  return <>
    <input ref={input} readOnly placeholder="YYYY/MM/DD" />
    <output>{date}</output>
  </>;
}
