کد استفاده در Alpine.js

با x-ref عنصر ورودی را به انتخابگر می‌دهیم و رویدادهای بومی کتابخانه مقدار x-model را همگام می‌کنند.

<input x-ref="date" x-model="date" readonly>
<output x-text="date"></output>

<script>
document.addEventListener('alpine:init', () => {
  Alpine.data('datePicker', () => ({
    date: '',
    init() {
      this.$nextTick(() => {
        this.picker = new JWheelPicker({ target: this.$refs.date });
      });
    },
    destroy() { this.picker?.destroy(); }
  }));
});
</script>

<main x-data="datePicker">...</main>
