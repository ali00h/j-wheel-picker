کد استفاده در Vue

نمونه بعد از mount شدن ورودی را به انتخابگر وصل می‌کند و هنگام خروج کامپوننت، منابع را آزاد می‌کند.

<script setup>
import { onMounted, onBeforeUnmount, ref } from 'vue';
const input = ref(null);
const date = ref('');
let picker;

onMounted(() => {
  picker = new JWheelPicker({
    target: input.value,
    onSelect: value => { date.value = value.formatted; }
  });
});
onBeforeUnmount(() => picker?.destroy());
</script>

<template>
  <input ref="input" readonly placeholder="YYYY/MM/DD">
  <output>{{ date }}</output>
</template>
